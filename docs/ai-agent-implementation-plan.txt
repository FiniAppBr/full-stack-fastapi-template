1.4 Database Models (Add to models.py)
```python
# Agent - AI agent configuration
class Agent(SQLModel, table=True):
    id: uuid.UUID = Field(default_factory=uuid.uuid4, primary_key=True)
    name: str
    model: str  # "gpt-4", "gpt-3.5-turbo", etc.
    temperature: float = 0.7
    system_prompt: str
    enabled_tools: list[str] = Field(default=[], sa_column=Column(JSON))
    max_tokens: int = 2000
    owner_id: uuid.UUID = Field(foreign_key="user.id")
    created_at: datetime

# Conversation - Chat sessions
class Conversation(SQLModel, table=True):
    id: uuid.UUID = Field(default_factory=uuid.uuid4, primary_key=True)
    agent_id: uuid.UUID = Field(foreign_key="agent.id")
    user_id: str  # External user (WhatsApp phone, etc.)
    title: str
    status: str  # "active", "archived", "pending"
    created_at: datetime
    last_message_at: datetime

# Message - Chat history
class Message(SQLModel, table=True):
    id: uuid.UUID = Field(default_factory=uuid.uuid4, primary_key=True)
    conversation_id: uuid.UUID = Field(foreign_key="conversation.id")
    role: str  # "user", "assistant", "system"
    content: str
    created_at: datetime

# AgentTool - Custom tools per agent
class AgentTool(SQLModel, table=True):
    id: uuid.UUID = Field(default_factory=uuid.uuid4, primary_key=True)
    agent_id: uuid.UUID = Field(foreign_key="agent.id")
    name: str
    description: str
    parameters: dict = Field(sa_column=Column(JSON))
    code: str  # Python function code
```
